# XHL Auto Loot v1.5.0
# The Native Sidecar skips the vanilla initial random delay of 0-5 seconds only for this mod's pickup areas.
# The range check runs about every 200 ms; an extra 123 ns in the private marker identifies this mod.
# Pickup continues when inventory slots are full if the item can still merge into an existing stack.
# Retries for that player and drop are paused only when vanilla rejects the item with InventoryFull.
enabled=true

# Regular gravity drops: twigs, stone, wood, ore fragments, etc.
collectGravityDrops=true

# Material piles: log piles, stone piles, resin piles, etc.
collectMaterialPiles=true

# Quest, collectible, and story items. Disabled by default, as in the original settings.
collectQuestItems=false

# Auto-pickup radius in meters. 5 meters can significantly reduce the number of range candidates.
pickupRadiusMeters=5

# Zero drop initial velocity, offset, and scatter radius based on the gathering node's actual components.
tuneGatherNodes=true

# Disable gravity on regular whitelisted drops to reduce bouncing and scattering.
softenGravityDrops=true

# Remove the spawn delay for whitelisted drops.
removeSpawnDelay=true

# Keep the flash indicator on ground loot.
keepLootSparks=true
