# XHL Auto Loot v1.5.0
# Native Sidecar skips the vanilla initial random wait of 0 to 5 seconds only for this mod's pickup areas.
# Range checks run about every 200 ms; the extra 123 ns private marker is used only to identify this mod.
# Pickup continues when inventory slots are full if the target item can still be merged into an existing stack.
# Retries for a player and a drop are paused only when vanilla determines the item cannot enter the inventory and raises InventoryFull.
enabled=true

# Ordinary gravity drops: branches, stone fragments, wood, ore chunks, etc.
collectGravityDrops=true

# Material piles: log piles, stone piles, resin piles, etc.
collectMaterialPiles=true

# Quest, collectible, and story items. Disabled by default, as in the original settings.
collectQuestItems=false

# Auto-pickup radius (meters); 5 meters can significantly reduce the number of range candidates.
pickupRadiusMeters=5

# Set drop initial velocity, offset, and scatter radius to zero based on the gather node's actual components.
tuneGatherNodes=true

# Disable gravity on ordinary whitelisted drops to reduce them bouncing away.
softenGravityDrops=true

# Remove the spawn delay for whitelisted drops.
removeSpawnDelay=true

# Keep the sparkle effect on ground pickups.
keepLootSparks=true
